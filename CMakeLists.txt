# LambdaEngine2D: an ECS 2D engine on raylib 6.0, C++20.
#   cmake --preset release && cmake --build --preset release && ctest --preset release -LE "gpu|bench"
cmake_minimum_required(VERSION 3.28)
if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Android" AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.3" CACHE STRING "Minimum macOS") # before project()
endif()
project(LambdaEngine2D VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(LAM_BUILD_TESTS "Tests" ${PROJECT_IS_TOP_LEVEL})
option(LAM_BUILD_EXAMPLES "Examples" ${PROJECT_IS_TOP_LEVEL})
option(LAM_BUILD_BENCH "Benchmarks" OFF)
option(LAM_BENCH_ENTT "EnTT baseline in lam_ecs_bench" OFF)
option(LAM_WERROR "Warnings are errors" OFF)
option(LAM_SANITIZE "ASan+UBSan" OFF)
option(LAM_LTO "LTO in Release" ON)
option(LAM_PROFILE "Profiler in Release builds" OFF)
include(cmake/LamDependencies.cmake)          # raylib, nlohmann_json, doctest (LAM_DEPS_DIR for offline)
include(cmake/LamHelpers.cmake)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/math/CMakeLists.txt)
  add_subdirectory(src/math)                    # lam_math
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/CMakeLists.txt)
  add_subdirectory(src/core)                    # lam_core only
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/ecs/CMakeLists.txt)
  add_subdirectory(src/ecs)                     # lam_ecs
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/app/CMakeLists.txt)
  add_subdirectory(src/core/app)                # lam_app (needs lam_ecs)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/backend/CMakeLists.txt)
  add_subdirectory(src/core/backend)            # lam_platform, lam_rl_common
endif()
set(LAM_FEATURE_MODULES assets render physics input audio scene tilemap animation particles ui)
foreach(m IN LISTS LAM_FEATURE_MODULES)         # dependency order
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/${m}/CMakeLists.txt)
    add_subdirectory(src/${m})
  endif()
endforeach()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/engine/CMakeLists.txt)
  add_subdirectory(src/core/engine)           # lam_engine: needs every module
endif()
if(LAM_BUILD_TESTS)
  enable_testing()
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/support/CMakeLists.txt)
    add_subdirectory(tests/support)
  endif()
  foreach(m IN ITEMS math core ecs ${LAM_FEATURE_MODULES})
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${m}/CMakeLists.txt)
      add_subdirectory(tests/${m})
    endif()
  endforeach()
endif()
if(LAM_BUILD_EXAMPLES)
  # Module demos (examples/basics/<nn>_<m>), small arcade games (examples/arcade/<nn>_<game>)
  # and the bigger example games (examples/games/<nn>_<game>).
  file(GLOB demos LIST_DIRECTORIES true ${CMAKE_CURRENT_SOURCE_DIR}/examples/basics/*
                                        ${CMAKE_CURRENT_SOURCE_DIR}/examples/arcade/*
                                        ${CMAKE_CURRENT_SOURCE_DIR}/examples/games/*)
  foreach(d IN LISTS demos)
    if(EXISTS ${d}/CMakeLists.txt)
      add_subdirectory(${d})
    endif()
  endforeach()
endif()
